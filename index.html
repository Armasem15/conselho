Instruções para Visual Studio Code (Node.js + MySQL)

Abaixo estão os scripts para conectar esta aplicação a um banco de dados real MySQL usando Node.js + Express no Visual Studio Code.

banco.sql 
CREATE DATABASE IF NOT EXISTS cef_grecia_d;
USE cef_grecia_d;

CREATE TABLE IF NOT EXISTS fichas_pre_conselho (
    id INT AUTO_INCREMENT PRIMARY KEY,
    professor VARCHAR(150) NOT NULL,
    componente VARCHAR(100) NOT NULL,
    turma VARCHAR(50) NOT NULL,
    data DATE NOT NULL,
    panorama_desenvolvimento TEXT NOT NULL,
    panorama_postura TEXT NOT NULL,
    aprendizagem_dificuldades TEXT NOT NULL,
    aprendizagem_frequencia TEXT NOT NULL,
    destaques_positivos TEXT NOT NULL,
    destaques_atencao TEXT NOT NULL,
    encaminhamento_estrategias TEXT NOT NULL,
    encaminhamento_obs TEXT,
    sintese_geral TEXT NOT NULL,
    
    -- Assinaturas Eletrônicas (Data + Confirmação)
    assinado_prof TINYINT(1) DEFAULT 1,
    data_assinado_prof DATETIME DEFAULT CURRENT_TIMESTAMP,
    
    assinado_pedagoga TINYINT(1) DEFAULT 0,
    nome_pedagoga VARCHAR(150) DEFAULT NULL,
    data_assinado_pedagoga DATETIME DEFAULT NULL,
    
    assinado_diretora TINYINT(1) DEFAULT 0,
    nome_diretora VARCHAR(150) DEFAULT NULL,
    data_assinado_diretora DATETIME DEFAULT NULL,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
server.js 
const express = require('express');
const mysql = require('mysql2');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());

// Conexão com Banco MySQL
const db = mysql.createPool({
    host: 'localhost',
    user: 'root',      // Altere para seu usuário
    password: '',      // Altere para sua senha
    database: 'cef_grecia_d',
    waitForConnections: true,
    connectionLimit: 10
});

// Listar todas as fichas
app.get('/api/fichas', (req, res) => {
    db.query('SELECT * FROM fichas_pre_conselho ORDER BY id DESC', (err, results) => {
        if (err) return res.status(500).json({ error: err.message });
        res.json(results);
    });
});

// Criar nova ficha
app.post('/api/fichas', (req, res) => {
    const data = req.body;
    const query = `INSERT INTO fichas_pre_conselho 
    (professor, componente, turma, data, panorama_desenvolvimento, panorama_postura, 
     aprendizagem_dificuldades, aprendizagem_frequencia, destaques_positivos, destaques_atencao, 
     encaminhamento_estrategias, encaminhamento_obs, sintese_geral) 
    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`;

    const values = [
        data.professor, data.componente, data.turma, data.data,
        data.panorama_desenvolvimento, data.panorama_postura,
        data.aprendizagem_dificuldades, data.aprendizagem_frequencia,
        data.destaques_positivos, data.destaques_atencao,
        data.encaminhamento_estrategias, data.encaminhamento_obs,
        data.sintese_geral
    ];

    db.query(query, values, (err, result) => {
        if (err) return res.status(500).json({ error: err.message });
        res.json({ success: true, id: result.insertId });
    });
});

// Assinar como Pedagoga ou Diretora
app.put('/api/fichas/:id/assinar', (req, res) => {
    const { id } = req.params;
    const { papel, nome } = req.body; // papel: 'pedagoga' ou 'diretora'

    let query = '';
    let params = [];

    if (papel === 'pedagoga') {
        query = 'UPDATE fichas_pre_conselho SET assinado_pedagoga = 1, nome_pedagoga = ?, data_assinado_pedagoga = NOW() WHERE id = ?';
        params = [nome, id];
    } else if (papel === 'diretora') {
        query = 'UPDATE fichas_pre_conselho SET assinado_diretora = 1, nome_diretora = ?, data_assinado_diretora = NOW() WHERE id = ?';
        params = [nome, id];
    }

    db.query(query, params, (err, result) => {
        if (err) return res.status(500).json({ error: err.message });
        res.json({ success: true });
    });
});

app.listen(3000, () => console.log('Servidor rodando em http://localhost:3000'));

Passo a Passo de Instalação no Visual Studio Code:
  1. Crie uma pasta no computador para seu projeto.
  2. Abra a pasta no Visual Studio Code.
  3. No terminal do VS Code, rode: npm init -y e depois npm install express mysql2 cors.
  4. Crie o arquivo server.js e cole o código Node acima.
  5. No seu gerenciador MySQL (XAMPP, MySQL Workbench, etc.), execute o SQL da aba 1.
  6. Inicie seu servidor Node com o comando: node server.js.
